import {
  useFrameFunction,
  useFrameFunctionMutation,
  useUserIdentity,
} from "@dust/react-hooks";
import { useState } from "react";

type ResponseList = {
  responses: { id: number; rating: number; comment: string; createdAt: string }[];
};

function describeError(err: unknown): string {
  if (
    err instanceof Error &&
    "code" in err &&
    err.code === "user_authentication_required"
  ) {
    return "Only the Frame author can read the responses.";
  }
  return err instanceof Error ? err.message : String(err);
}

function Responses() {
  const list = useFrameFunction("list-responses", {});
  const resetDemoData = useFrameFunctionMutation("reset-demo-data");
  const payload = list.data as ResponseList | undefined;

  return (
    <section className="mt-8">
      <div className="mb-2 flex items-center gap-4">
        <h2 className="text-xl font-semibold">Responses (author only)</h2>
        <button
          onClick={async () => {
            await resetDemoData.trigger({});
            await list.mutate();
          }}
          className="text-sm text-muted-foreground underline"
        >
          Reset sample data
        </button>
      </div>
      {list.isLoading && <p className="text-muted-foreground">Loading...</p>}
      {list.error && <p className="text-red-600">{describeError(list.error)}</p>}
      <ul className="space-y-2">
        {(payload?.responses ?? []).map((response) => (
          <li key={response.id} className="rounded-md border border-border p-3">
            <p className="font-medium">{response.rating} / 5</p>
            <p className="text-sm">{response.comment}</p>
          </li>
        ))}
      </ul>
    </section>
  );
}

export default function App() {
  const [rating, setRating] = useState(5);
  const [comment, setComment] = useState("");
  const [status, setStatus] = useState<string | null>(null);
  const identity = useUserIdentity();
  const submit = useFrameFunctionMutation("submit-response");

  async function handleSubmit() {
    setStatus(null);
    try {
      await submit.trigger({ rating, comment });
      setComment("");
      setStatus("Thanks, your response was recorded.");
    } catch (err) {
      setStatus(describeError(err));
    }
  }

  return (
    <div className="min-h-screen bg-background p-8 text-foreground">
      <h1 className="mb-2 text-3xl font-bold">Feedback</h1>
      <p className="mb-6 text-sm text-muted-foreground">
        Anyone with the link can submit a response. Only the Frame author can
        read them.
      </p>
      <div className="mb-4 flex gap-2">
        {[1, 2, 3, 4, 5].map((value) => (
          <button
            key={value}
            onClick={() => setRating(value)}
            className={
              value === rating
                ? "rounded-md bg-primary px-3 py-1 text-primary-foreground"
                : "rounded-md border border-border px-3 py-1"
            }
          >
            {value}
          </button>
        ))}
      </div>
      <textarea
        value={comment}
        onChange={(event) => setComment(event.target.value)}
        placeholder="Your comment"
        className="mb-4 w-full rounded-md border border-border px-3 py-2"
      />
      <button
        onClick={handleSubmit}
        disabled={submit.isMutating}
        className="rounded-md bg-primary px-4 py-2 text-primary-foreground"
      >
        Submit
      </button>
      {status && <p className="mt-4 text-sm">{status}</p>}
      {identity.isFrameAuthor && <Responses />}
    </div>
  );
}
